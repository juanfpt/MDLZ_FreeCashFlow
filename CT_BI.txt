let
    // HELPER FUNCTIONS-----------------------------------------------------------------------------------------
    
    fnGetUsername = () =>
        let
            Source = Folder.Contents("C:\Users\"),
            #"Expanded Attributes" = Table. ExpandRecordColumn(Source, "Attributes", 
                {"Hidden", "Directory", "ChangeTime"}, 
                {"Hidden", "Directory", "ChangeTime"}
            ),
            #"Filtered Directories not hidden" = Table.SelectRows(#"Expanded Attributes", 
                each [Directory] = true and [Hidden] = false
            ),
            #"Removed Errors" = Table.RemoveRowsWithErrors(#"Filtered Directories not hidden", {"ChangeTime"}),
            #"Filtered Rows" = Table.SelectRows(#"Removed Errors", each [Name] <> "Public"),
            Username = #"Filtered Rows"{0}[Name]
        in
            Username,
    
    fnGetBasePath = (username as text) as text =>
        "C:\Users\" & username & "\MDLZ\Cash Flow Planning WACAM - Documents\Cash Flow Mensual\FCF Internal Report\",
    
    // Función para convertir nombre de mes en inglés a número
    fnMonthNameToNumber = (monthName as text) as number =>
        let
            MonthMap = #table(
                {"Month", "Number"},
                {
                    {"January", 1}, {"February", 2}, {"March", 3}, {"April", 4},
                    {"May", 5}, {"June", 6}, {"July", 7}, {"August", 8},
                    {"September", 9}, {"October", 10}, {"November", 11}, {"December", 12}
                }
            ),
            Result = try 
                MonthMap{[Month = monthName]}[Number]
            otherwise
                null
        in
            Result,
    
    // CONFIGURATION--------------------------------------------------------------------------------------------
    
    Username = fnGetUsername(),
    BasePath = fnGetBasePath(Username),
    BIFolderPath = BasePath & "Data Repository\Actuals\3.   Cash Taxes\BI",
    
    // LOAD FILES FROM FOLDER-----------------------------------------------------------------------------------
    
    SourceFiles = Folder.Files(BIFolderPath),
    
    // Filter hidden files and non-Excel files
    #"Filtered Hidden Files" = Table.SelectRows(SourceFiles, 
        each [Attributes]? [Hidden]?  <> true
    ),
    
    #"Filtered Excel Files" = Table.SelectRows(#"Filtered Hidden Files",
        each Text.EndsWith([Name], ".xlsx") or Text.EndsWith([Name], ".xls")
    ),
    
    // Extract Year from filename (assumes filename is like "2024. xlsx" or "2025.xlsx")
    #"Added Year from Filename" = Table.AddColumn(#"Filtered Excel Files", "Year", 
        each 
            let
                FileName = Text.BeforeDelimiter([Name], ". "),  // Obtiene "2024" de "2024.xlsx"
                YearNumber = try Number.From(FileName) otherwise null
            in
                YearNumber,
        Int64.Type
    ),
    
    // Load Excel content
    #"Added Excel Content" = Table.AddColumn(#"Added Year from Filename", "ExcelData", 
        each try Excel. Workbook([Content], null, true) otherwise null
    ),
    
    // Filter successfully loaded files
    #"Filtered Loaded Files" = Table.SelectRows(#"Added Excel Content", 
        each [ExcelData] <> null
    ),
    
    // Expand to get sheets
    #"Expanded Excel Sheets" = Table.ExpandTableColumn(#"Filtered Loaded Files", "ExcelData", 
        {"Name", "Data", "Kind"}, 
        {"SheetName", "SheetData", "ItemKind"}
    ),
    
    // Filter only sheets (not tables or named ranges)
    #"Filtered Sheets Only" = Table.SelectRows(#"Expanded Excel Sheets", 
        each [ItemKind] = "Sheet"
    ),
    
    // Promote headers
    #"Promoted Headers in Sheets" = Table.AddColumn(#"Filtered Sheets Only", "DataWithHeaders",
        each Table.PromoteHeaders([SheetData], [PromoteAllScalars=true])
    ),
    
    // Select only needed columns before combining
    #"Select Relevant Columns" = Table.SelectColumns(#"Promoted Headers in Sheets",
        {"Year", "DataWithHeaders"}
    ),
    
    // Add Year column to each sheet's data before combining
    #"Added Year to Data" = Table.AddColumn(#"Select Relevant Columns", "DataWithYear",
        each Table.AddColumn([DataWithHeaders], "FileYear", each [Year], Int64.Type)
    ),
    
    // Combine all sheets from all files
    #"Combined All Sheets" = Table.Combine(#"Added Year to Data"[DataWithYear]),
    
    // DATA TYPE TRANSFORMATIONS--------------------------------------------------------------------------------
    
    #"Changed Initial Types" = Table.TransformColumnTypes(#"Combined All Sheets",
        {
            {"Country", type text},
            {"Description", type text},
            {"Month", type text},
            {"CT.  Actual", type number},
            {"FileYear", Int64.Type}
        }
    ),
    
    // ADD KEY DATE COLUMN (Year + Month)-----------------------------------------------------------------------
    
    #"Added Month Number" = Table.AddColumn(#"Changed Initial Types", "MonthNumber",
        each fnMonthNameToNumber([Month]),
        Int64.Type
    ),
    
    // Create Key Date as end of month
    #"Added Key Date" = Table.AddColumn(#"Added Month Number", "Key Date",
        each Date.EndOfMonth(#date([FileYear], [MonthNumber], 1)),
        type date
    ),
    
    // REMOVE UNNECESSARY COLUMNS-------------------------------------------------------------------------------
    
    #"Removed Month and Year Columns" = Table.RemoveColumns(#"Added Key Date",
        {"Month", "FileYear", "MonthNumber"}
    ),
    
    // REORDER COLUMNS------------------------------------------------------------------------------------------
    
    #"Reordered Columns" = Table.ReorderColumns(#"Removed Month and Year Columns",
        {"Country", "Description", "Key Date", "CT. Actual"}
    ),
    
    // RENAME COLUMNS (opcional)--------------------------------------------------------------------------------
    
    #"Renamed Columns" = Table.RenameColumns(#"Reordered Columns",
        {{"CT. Actual", "Value"}}
    ),
    
    // FINAL RESULT---------------------------------------------------------------------------------------------
    
    Result = #"Renamed Columns"
in
    Result
