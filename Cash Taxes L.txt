let

    // HELPER FUNCTIONS-------------------------------------------------------------------------------------------------
    
    fnGetUsername = () =>
        let
            Source = Folder.Contents("C:\Users\"),
            #"Expanded Attributes" = Table.ExpandRecordColumn(Source, "Attributes", 
                {"Hidden", "Directory", "ChangeTime"}, 
                {"Hidden", "Directory", "ChangeTime"}
            ),
            #"Filtered Directories not hidden" = Table.SelectRows(#"Expanded Attributes", 
                each [Directory] = true and [Hidden] = false
            ),
            #"Removed Errors" = Table.RemoveRowsWithErrors(#"Filtered Directories not hidden", {"ChangeTime"}),
            #"Filtered Rows" = Table.SelectRows(#"Removed Errors", each [Name] <> "Public"),
            Username = #"Filtered Rows"{0}[Name]
        in
            Username,
    
    fnGetBasePath = (username as text) as text =>
        "C:\Users\" & username & "\MDLZ\Cash Flow Planning WACAM - Documents\Cash Flow Mensual\FCF Internal Report\",
    
    fnCountryMatches = (ruleCountry as text, actualCountry as text) as logical =>
        let
            RuleCountryUpper = Text.Upper(ruleCountry),
            ActualCountryUpper = Text.Upper(actualCountry),
            IsAll = RuleCountryUpper = "ALL",
            CountryList = List.Transform(Text.Split(RuleCountryUpper, ","), each Text.Trim(_)),
            IsInList = List.Contains(CountryList, ActualCountryUpper),
            Matches = IsAll or IsInList
        in
            Matches,
    
    // CONFIGURATION----------------------------------------------------------------------------------------------------
    
    Username = fnGetUsername(),
    BasePath = fnGetBasePath(Username),
    
    ClassificationFilePath = BasePath & "Data Repository\Actuals\Support\Classifier.xlsx",


    // LOAD CLASSIFICATION RULES FROM EXTERNAL EXCEL FILE--------------------------------------------------------------

    
    ClassificationWorkbook = Excel.Workbook(File.Contents(ClassificationFilePath), null, true),
    ClassificationSheet = ClassificationWorkbook{[Item="TaxClassification",Kind="Table"]}[Data],
    
    ClassificationRules = Table.TransformColumnTypes(ClassificationSheet,
        {
            {"Country", type text},
            {"Keyword", type text},
            {"Category", type text},
            {"Priority", Int64.Type}
        }
    ),
    
    ClassificationRulesUpper = Table.TransformColumns(ClassificationRules,
        {
            {"Keyword", Text.Upper, type text},
            {"Country", Text.Upper, type text}
        }
    ),
    
    ClassificationRulesSorted = Table.Sort(ClassificationRulesUpper, {{"Priority", Order.Ascending}}),
    

    // CLASSIFICATION FUNCTION (WITH COUNTRY FILTER AND PRIORITY)----------------------------------------------------

    
    fnClassifyTransaction = (textToClassify as text, country as text, classificationRules as table) as text =>
        let
            TextUpper = Text.Upper(textToClassify),
            
            ApplicableRules = Table.SelectRows(classificationRules, 
                each fnCountryMatches([Country], country)
            ),
            
            CountrySpecificRules = Table.SelectRows(ApplicableRules, 
                each Text.Upper([Country]) = Text.Upper(country)
            ),
            
            MultiCountryRules = Table.SelectRows(ApplicableRules, 
                each Text.Contains([Country], ",")
            ),
            
            GlobalRules = Table.SelectRows(ApplicableRules, 
                each Text.Upper([Country]) = "ALL"
            ),
            
            // Match function by priority
            FindMatch = (rules as table) as nullable text =>
                let
                    // Buscar la primera keyword que coincida (orden por Priority)
                    MatchingRows = Table.SelectRows(rules, 
                        each Text.Contains(TextUpper, [Keyword])
                    ),
                    
                    // Retornar la categoría de la primera coincidencia (mayor prioridad)
                    FirstMatch = if Table.RowCount(MatchingRows) > 0 
                                then MatchingRows{0}[Category] 
                                else null
                in
                    FirstMatch,
            
            // Buscar en orden: país específico > multi-país > global
            CountryMatch = FindMatch(CountrySpecificRules),
            MultiCountryMatch = FindMatch(MultiCountryRules),
            GlobalMatch = FindMatch(GlobalRules),
            
            FinalCategory = if CountryMatch <> null then CountryMatch 
                            else if MultiCountryMatch <> null then MultiCountryMatch 
                            else if GlobalMatch <> null then GlobalMatch 
                            else "Others"
        in
            FinalCategory,
    

    // DATA FOLDER PATH-------------------------------------------------------------------------------------------------

    
    DataFolderPath = BasePath & "Data Repository\Actuals\3. Cash Taxes",
    

    // LOAD FILES FROM FOLDER

    
    SourceFiles = Folder.Files(DataFolderPath),
    #"Filtered Hidden Files" = Table.SelectRows(SourceFiles, 
        each [Attributes]? [Hidden]?  <> true
    ),
    
    // TRANSFORM FILES-------------------------------------------------------------------------------------------------

    
    #"Invoked Transform Function" = Table.AddColumn(#"Filtered Hidden Files", "Transform File", 
        each #"Transform File"([Content])
    ),
    #"Renamed Source Column" = Table.RenameColumns(#"Invoked Transform Function", {{"Name", "Source.Name"}}),
    #"Selected Relevant Columns" = Table.SelectColumns(#"Renamed Source Column", {"Source.Name", "Transform File"}),
    #"Expanded Transformed Data" = Table.ExpandTableColumn(
        #"Selected Relevant Columns", 
        "Transform File", 
        Table.ColumnNames(#"Transform File"(#"Sample File"))
    ),
    
    // DATA TYPE TRANSFORMATIONS----------------------------------------------------------------------------------------
    
    #"Changed Initial Types" = Table.TransformColumnTypes(#"Expanded Transformed Data", 
        {
            {"Source.Name", type text}, 
            {"Company Code", type text}, 
            {"Account", type text}, 
            {"Document Type", type text}, 
            {"Document Date", type date},
            {"Posting Date", type date},
            {"Year/month", type date},
            {"Text", type text},
            {"Amount in local currency", type number}
        }
    ),
    
    // FILTER UNWANTED DOCUMENT TYPES-----------------------------------------------------------------------------------
    
    #"Filtered Document Types" = Table.SelectRows(#"Changed Initial Types", 
        each not List.Contains({"AB", "MA", "MB"}, [Document Type])
    ),
    
    // CLEAN UP COLUMNS------------------------------------------------------------------------------------------------
    
    #"Removed Source Name" = Table.RemoveColumns(#"Filtered Document Types", {"Source.Name"}),
    #"Removed Date Columns" = Table.RemoveColumns(#"Removed Source Name", {"Document Date", "Posting Date"}),
    

    // ENRICH WITH COUNTRY DATA (BEFORE CLASSIFICATION)----------------------------------------------------------------
    
    #"Merged with Countries" = Table.NestedJoin(
        #"Removed Date Columns", 
        {"Company Code"}, 
        Countries_SAP, 
        {"Entity"}, 
        "Countries_SAP", 
        JoinKind.LeftOuter
    ),
    #"Expanded Country Column" = Table.ExpandTableColumn(
        #"Merged with Countries", 
        "Countries_SAP", 
        {"Country"}, 
        {"Country"}
    ),
    
    
    #"Added Description Category" = Table.AddColumn(#"Expanded Country Column", "Description", 
        each fnClassifyTransaction([Text], [Country]),
        type text
    ),
    
    #"Renamed Date Column" = Table.RenameColumns(#"Added Description Category", {{"Year/month", "Key Date"}}),
    
    #"Added FX Lookup Key" = Table.AddColumn(#"Renamed Date Column", "ID", 
        each [Country] & 
             Date.MonthName([Key Date]) & 
             Number.ToText(Date.Year([Key Date])) & 
             "Plrate" & 
             "ACT"
    ),
    
    #"Merged with FX Rates" = Table.NestedJoin(
        #"Added FX Lookup Key", 
        {"ID"}, 
        #"Foreign_Exchange [Master]", 
        {"ID"}, 
        "Foreign_Exchange [Master]", 
        JoinKind.LeftOuter
    ),
    #"Expanded FX Value" = Table.ExpandTableColumn(
        #"Merged with FX Rates", 
        "Foreign_Exchange [Master]", 
        {"Value"}, 
        {"FX Rate"}
    ),
    
    #"Renamed Amount Column" = Table.RenameColumns(#"Expanded FX Value", 
        {{"Amount in local currency", "Amount LC"}}
    ),
    #"Added USD Value" = Table.AddColumn(#"Renamed Amount Column", "Value", 
        each ([Amount LC] / [FX Rate]) / 1000000
    ),
    #"Set Value Type" = Table.TransformColumnTypes(#"Added USD Value", {{"Value", type number}}),
    
    #"Removed Temporary Columns" = Table.RemoveColumns(#"Set Value Type", {"ID", "FX Rate"}),
    
    Result = #"Removed Temporary Columns"
in
    Result
